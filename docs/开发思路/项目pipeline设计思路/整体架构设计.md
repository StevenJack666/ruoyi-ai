
整体设计思路使用pipeline设计：
用户输入
  ├→ 内容审核（有无违规？）
  ├→ RAG 检索（知识库相关？）
  ├→ 工具调用（需要查数据库/搜网页？）
  ├→ 多模态（上传了图片？）
  ├→ 记忆管理（历史会话？）
  └→ AI 模型（最终生成回复）


Pipeline 方案
// 定义 Filter 链
ChatPipeline pipeline = ChatPipeline.builder()
    .add(new ContentModerationFilter())    // 1. 内容审核
    .add(new RagRetrievalFilter())         // 2. RAG 检索
    .add(new ToolExecutionFilter())        // 3. 工具调度
    .add(new StreamingChatFilter())        // 4. AI 生成
    .add(new ResponseModerationFilter())   // 5. 响应审核
    .build();

// 执行
ChatResponse response = pipeline.execute(request);


每个 Filter 独立
// 内容审核 Filter — 违规直接阻断
public class ContentModerationFilter implements ChatFilter {
    @Override
    public void doFilter(ChatRequest request, ChatFilterChain chain) {
        if (containsViolation(request.getContent())) {
            return error("内容违规，已拦截");  // 直接返回，不继续
        }
        chain.doFilter(request);  // 放行
    }
}

// RAG Filter — 检索知识库，结果拼入上下文
public class RagRetrievalFilter implements ChatFilter {
    @Override
    public void doFilter(ChatRequest request, ChatFilterChain chain) {
        List<String> docs = knowledgeRetrievalService.retrieveTexts(request);
        request.setContextDocuments(docs);  // 增强上下文
        chain.doFilter(request);
    }
}
